</div>
<div class="story-body">
<p>Kanda was born from a simple idea: cooking shouldn't have to be complicated to taste good.</p>
<p>We noticed how often great meals require a long list of individual spices — measuring, searching through cupboards, wondering whether you have everything you need, and trying to get the proportions right. Yet sometimes, all you really want is to make a delicious meal without all the fuss.</p>
<p>So Kanda was created to bring those ingredients together in carefully balanced, ready-to-use blends, taking away the measuring and guesswork while keeping the flavour.</p>
<p>Our first product, <em>Chicken Rub</em>, was created with exactly that in mind: a blend that makes preparing well-seasoned chicken simple and effortless.</p>
<p>But Kanda isn't just about chicken. It's about creating a growing collection of blends for the foods people actually love to cook; whether that's crispy fries, juicy chicken, flavourful beef, smoky BBQ, or a fragrant Kenyan pilau.</p>
<p>Kanda is about making everyday cooking feel a little easier, while making every meal taste a little better. And that's where <em>"Effortlessly Good"</em> comes from.</p>

return Object.values(skewers).reduce((s,q) => s+q, 0);
}

function updateSkewerPriceLabels(){
const total = totalSkewerQty();
const unit = skewerUnitPrice(total);
document.querySelectorAll('.skewer-price').forEach(el => {
el.textContent = `KES ${unit} / stick`;
});
}

document.querySelectorAll('.skewer-stepper').forEach(stepper => {
const item = stepper.closest('.item');
const flavor = item.dataset.flavor;
const qtyEl = stepper.querySelector('.qty');
stepper.querySelector('.inc').addEventListener('click', () => {
skewers[flavor] = (skewers[flavor] || 0) + 1;
qtyEl.textContent = skewers[flavor];
updateSkewerPriceLabels();
renderCart();
});
stepper.querySelector('.dec').addEventListener('click', () => {
if(!skewers[flavor]) return;
skewers[flavor] = Math.max(0, skewers[flavor] - 1);
qtyEl.textContent = skewers[flavor];
updateSkewerPriceLabels();
renderCart();
});
});

document.querySelectorAll('.jar-stepper').forEach(stepper => {
const item = stepper.closest('.item');
const jarName = item.dataset.jar;
const price = parseInt(item.dataset.price, 10);
const qtyEl = stepper.querySelector('.qty');
stepper.querySelector('.inc').addEventListener('click', () => {
if(!jars[jarName]) jars[jarName] = { qty: 0, price };
jars[jarName].qty += 1;
qtyEl.textContent = jars[jarName].qty;
renderCart();
});
stepper.querySelector('.dec').addEventListener('click', () => {
if(!jars[jarName] || jars[jarName].qty === 0) return;
jars[jarName].qty -= 1;
qtyEl.textContent = jars[jarName].qty;
renderCart();
});
});

function removeSkewer(flavor){
skewers[flavor] = 0;
document.querySelectorAll('.item').forEach(item => {
if(item.dataset.flavor === flavor){
item.querySelector('.qty').textContent = '0';
}
});
updateSkewerPriceLabels();
renderCart();
}

function removeJar(jarName){
if(jars[jarName]) jars[jarName].qty = 0;
document.querySelectorAll('.item').forEach(item => {
if(item.dataset.jar === jarName){
item.querySelector('.qty').textContent = '0';
}
});
renderCart();
}
window.removeSkewer = removeSkewer;
window.removeJar = removeJar;

function changeSkewer(flavor, delta){
skewers[flavor] = Math.max(0, (skewers[flavor] || 0) + delta);
document.querySelectorAll('.item').forEach(item => {
if(item.dataset.flavor === flavor) item.querySelector('.qty').textContent = skewers[flavor];
});
updateSkewerPriceLabels();
renderCart();
}

function changeJar(jarName, delta){
if(!jars[jarName]) return;
jars[jarName].qty = Math.max(0, jars[jarName].qty + delta);
document.querySelectorAll('.item').forEach(item => {
if(item.dataset.jar === jarName) item.querySelector('.qty').textContent = jars[jarName].qty;
});
renderCart();
}

function renderCart(){
const list = document.getElementById('cart-list');
const totalEl = document.getElementById('cart-total');
const depositBox = document.getElementById('deposit-box');
const depositAmountEl = document.getElementById('deposit-amount');
const balanceAmountEl = document.getElementById('balance-amount');

const skewerEntries = Object.entries(skewers).filter(([,q]) => q > 0);
const jarEntries = Object.entries(jars).filter(([,v]) => v.qty > 0);

if(skewerEntries.length === 0 && jarEntries.length === 0){
list.innerHTML = '<p class="empty-note">Nothing added yet — pick items from the menu above.</p>';
totalEl.textContent = 'KES 0';
depositBox.style.display = 'none';
return;
}

const totalSkewers = totalSkewerQty();
const unitPrice = skewerUnitPrice(totalSkewers);
let skewerSubtotal = 0;
let rows = '';

skewerEntries.forEach(([flavor, qty]) => {
const lineTotal = qty * unitPrice;
skewerSubtotal += lineTotal;
rows += `<div class="cart-row"><span>${flavor}<small style="display:block;opacity:.65;">KES ${unitPrice} each</small></span><span>KES ${lineTotal}</span><span class="cart-controls"><button type="button" onclick="changeSkewer('${flavor.replace(/'/g,"\\'")}',-1)">−</button><span class="cart-qty">${qty}</span><button type="button" onclick="changeSkewer('${flavor.replace(/'/g,"\\'")}',1)">+</button></span></div>`;
});

let jarSubtotal = 0;
jarEntries.forEach(([jarName, v]) => {
const lineTotal = v.qty * v.price;
jarSubtotal += lineTotal;
rows += `<div class="cart-row"><span>${jarName}</span><span>KES ${lineTotal}</span><span class="cart-controls"><button type="button" onclick="changeJar('${jarName.replace(/'/g,"\\'")}',-1)">−</button><span class="cart-qty">${v.qty}</span><button type="button" onclick="changeJar('${jarName.replace(/'/g,"\\'")}',1)">+</button></span></div>`;
});

list.innerHTML = rows;

const total = skewerSubtotal + jarSubtotal;
totalEl.textContent = 'KES ' + total;

if(skewerSubtotal > 0){
const deposit = Math.round(skewerSubtotal / 2) + jarSubtotal;
const balance = skewerSubtotal - Math.round(skewerSubtotal / 2);
depositAmountEl.textContent = 'KES ' + deposit;
balanceAmountEl.textContent = 'KES ' + balance;
depositBox.style.display = 'block';
} else {
depositBox.style.display = 'none';
}
}
